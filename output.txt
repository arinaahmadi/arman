from django.shortcuts import get_object_or_404, redirect, render

from .forms import ReceiverForm
from .models import Receiver


def listreceiver_view(request):
    receivers = Receiver.objects.all()
    return render(request, 'managebar/receivers.html', {'receivers': receivers})


def detailreceiver_view(request, pk):
    receiver = get_object_or_404(Receiver, pk=pk)
    return render(request, 'managebar/receiver.html', {'receiver': receiver})


def createreceiver_view(request):
    if request.method == 'POST':
        form = ReceiverForm(request.POST)
        if form.is_valid():
            form.save()
            return redirect('managebar:receivers')
        else:
            print(form.errors)   # برای دیباگ
    else:
        form = ReceiverForm()

    return render(request, 'managebar/createreceiver.html', {'form': form})


def updatereceiver_view(request, pk):
    receiver = get_object_or_404(Receiver, pk=pk)

    if request.method == 'POST':
        form = ReceiverForm(request.POST, instance=receiver)
        if form.is_valid():
            form.save()
            return redirect('managebar:receivers')
        else:

    else:

def deletereceiver_view(request, pk):
    receiver = get_object_or_404(Receiver, pk=pk)
